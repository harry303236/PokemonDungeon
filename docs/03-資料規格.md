# 03 資料規格

所有 runtime 資料都放在 `src/data/generated/*.json`，由 `scripts/build-data.ts` 產生。下列 TS 介面同時要寫成 zod schema（`src/data/schemas.ts`），用 `z.infer` 推導型別。id 一律 kebab-case（與 PokeAPI 一致）。

## 1. 共用型別
```ts
export type TypeId = 'normal'|'fire'|'water'|'grass'|'electric'|'ice'|'fighting'|'poison'|'ground'|'flying'|'psychic'|'bug'|'rock'|'ghost'|'dragon'|'dark'|'steel'|'fairy';
export type StatId = 'hp'|'attack'|'defense'|'special-attack'|'special-defense'|'speed';
export type DamageClass = 'physical'|'special'|'status';
export type WeatherId = 'none'|'sun'|'rain'|'sand'|'snow';
export type TerrainId = 'none'|'grassy'|'fire-field'|'mud'|'steam';   // 遊戲自訂地帶，不是正作場地
export type StatusId = 'burn'|'poison'|'toxic'|'paralysis'|'sleep'|'freeze'|'confusion'|'flinch'|'slow'|'blind'|'fear'|'bound';
export type TagId = 'wet'|'scorched'|'seeded'|'marked'|'heat'|string; // 詞綴可宣告新 tag
export type Rarity = 'common'|'uncommon'|'rare'|'epic'|'legendary';
export interface LocalizedName { 'zh-Hant': string; en: string; }
```

## 2. Pokemon（species）
```ts
export interface PokemonSpecies {
  id: string;                 // 'charmander'
  dexNo: number;              // 4
  name: LocalizedName;        // 來自 PokeAPI species.names
  types: TypeId[];
  baseStats: Record<StatId, number>;
  abilities: { id: string; hidden: boolean }[];
  evolutionLine: string;      // 'charmander-line'
  evolvesTo?: { speciesId: string; minLevel: number };
  learnset: { moveId: string; method: 'level-up'|'machine'|'egg'|'tutor'|'evolution'; level: number }[]; // scarlet-violet；level 0 = 進化時習得（method 改寫為 'evolution'）
  spriteKey: string;          // 對應 sprite-provider，例如 'pkmn/charmander'
  radius: number;             // 碰撞半徑 px，設計值（預設 14，最終型 18，Boss 另設）
}
```

## 3. Move
```ts
export interface MoveData {           // 來自 API 的事實
  id: string; name: LocalizedName; type: TypeId; damageClass: DamageClass;
  power: number|null; accuracy: number|null; pp: number; priority: number;
  effectChance: number|null; shortEffect: string; // en

}
export interface MoveBehavior {       // data/design/move-behaviors.json，手寫的設計改編
  moveId: string;
  shape: ShapeSpec;
  castTime: number;          // 秒，引導／蓄力
  recovery: number;          // 施放後硬直
  energyCostOverride?: number; // 不填 = 依公式 04 §5.2

  selfEffects?: EffectSpec[];

  sfx?: string;

export type ShapeSpec =

 | { kind:'aoe-self'; radius:number }
```

## 5. Item（攜帶道具與 TM）
```ts
export interface ItemBase {
  id: string; name: LocalizedName; kind: 'held'|'tm'|'material'|'currency';
  iconKey: string; stackable: boolean;
  baseEffects?: EffectSpec[];     // held：例如 charcoal 火招 +20%
  tmMoveId?: string;              // kind==='tm'
  dropLevelMin: number;
}
export interface ItemInstance {   // 存檔用
  uid: string;                    // nanoid
  baseId: string; rarity: Rarity; itemLevel: number;
  affixes: { affixId: string; rolls: number[] }[];  // rolls 對應 AffixData.params 的實際擲出值
  locked?: boolean;
}
```
MVP 攜帶道具：`charcoal`（火招+20%）、`shell-bell`（造成傷害的 8% 回血）、`focus-band`（致命傷 10% 機率留 1 HP，60 秒內最多一次）、`quick-claw`（冷卻 -10%）、`expert-belt`（效果絕佳 +20%）、`life-orb`（傷害 +30%，每次施招扣 3% HP）。

## 6. Affix
```ts
export type AffixCategory = 'stat'|'shape'|'trigger'|'reaction'|'cost';
export interface AffixData {
  id: string;                        // 'split-flame'
  name: LocalizedName; category: AffixCategory;
  slot: ('tm'|'held')[];
  rarityMin: Rarity; weight: number;
  requires?: { moveIds?: string[]; moveTypes?: TypeId[]; damageClass?: DamageClass[]; shapeKinds?: ShapeSpec['kind'][]; lines?: string[] };
  params: { key: string; min: number; max: number; step?: number }[];
  effects: EffectSpec[];             // 用 {param:key} 參照 rolls
  descTemplate: string;              // i18n key，例如 'affix.split-flame.desc'，內含 {count}
  tagsProduced?: TagId[]; tagsConsumed?: TagId[];
}
export type EffectSpec =
 | { op:'modifyShape'; set: Partial<Record<string, number|string|{param:string}>> ; kindTo?: ShapeSpec['kind'] }
 | { op:'statMod'; stat: string; mode:'add'|'mul'; value:number|{param:string}; condition?: Condition }
 | { op:'listen'; event: keyof GameEventsMap; condition?: Condition; actions: ActionSpec[]; icd?: number } // icd 內部冷卻秒數
 | { op:'replaceCost'; from:'hpTick'|'pp'|'energy'; to: ActionSpec };
export type ActionSpec =
 | { do:'applyStatus'; status:StatusId; chance:number|{param:string}; duration:number; target:'hitTarget'|'self'|'area'; radius?:number }
 | { do:'applyTag'; tag:TagId; duration:number; stacks?:number; target:'hitTarget'|'self' }
 | { do:'consumeTag'; tag:TagId; target:'hitTarget' }
 | { do:'spawnExplosion'; radius:number; powerPct:number|{param:string}; type?:TypeId; vfx:string }
 | { do:'spawnZone'; terrain:TerrainId; radius:number; duration:number }
 | { do:'castMove'; moveId:string; powerPct:number; from:'hitTarget'|'self' }
 | { do:'heal'; pct:number; of:'maxHp'|'damageDealt' }
 | { do:'damageSelf'; pct:number }
 | { do:'addShield'; pct:number }
 | { do:'reduceCooldown'; slot:'this'|'all'; seconds:number }
 | { do:'grantStack'; stack:string; max:number };
export interface Condition { all?: Condition[]; any?: Condition[]; weather?: WeatherId[]; targetHasTag?: TagId; targetHasStatus?: StatusId; hpBelowPct?: number; moveType?: TypeId[]; moveId?: string[]; isCrit?: boolean; killedBy?: 'burn'|'poison'|'any' }
```

## 7. SkillTree
```ts
export interface SkillTree {
  lineId: string;                       // 'charmander-line'
  branches: { id: string; name: LocalizedName; color: string }[];  // 3 個
  tiers: { tier: 1|2|3|4; requiresSpeciesId: string; minLevel: number }[];
  // 1=基礎型 Lv1、2=中間型 Lv16、3=最終型 Lv36(Bulbasaur 32)、4=頂層 Lv50
  nodes: SkillNode[];
}
export interface SkillNode {
  id: string; branch: string; tier: 1|2|3|4;
  kind: 'move'|'passive'|'keystone';
  moveId?: string;                     // kind==='move' 必須在該線 learnset 中（build-data 驗證）
  maxRank: number;                     // move:3（每級 +10% power）；passive:1–5；keystone:1
  cost: number;                        // 每 rank 點數
  requires: string[];                  // 前置節點 id（要求 rank ≥ 1）
  effects: EffectSpec[];               // passive／keystone 的效果，per rank
  pos: { x: number; y: number };      // UI 格子座標
}
```
規則：同 tier 的節點，需要在該分支已投入 `tier*3` 點以上才能點（tier1 不需要）。
`charmander-line` 完整節點表：依 `08` §1.3 轉寫，id 前綴 `cl-`（例：`cl-a-flamethrower`）。

## 12. 資料管線（scripts）
### `fetch-pokeapi.ts`
- 輸入：`data/design/species-list.json`（MVP：charmander 線＋10 種怪＋onix；同時抓 squirtle、bulbasaur 線）。
- 對每個 species：GET `pokemon/{id}`、`pokemon-species/{id}`、`evolution-chain`；收集出現過的 move、ability，再 GET `move/{id}`、`ability/{id}`、`type/{id}`（18 種，組屬性表）。
- 快取：`data/raw/{endpoint}/{id}.json`；已存在就跳過（`--force` 重抓）。併發 ≤ 8，失敗重試 3 次、指數退避；要帶 `User-Agent: pokedungeon-data-script`。
- 名稱：`names` 取 `zh-hant`（API 的 language code 是 `zh-hant`），沒有就用 `en`。
- learnset：只取 `version_group = scarlet-violet`；沒有就退回該寶可夢的最後一個 version group，並在報告中輸出警告。（注意：不能用 version group id 最大值，因為會選到 `blue-japan`／`champions` 這類不完整的資料。）
### `build-data.ts`
- 合併 raw＋`data/design/*` → `src/data/generated/{species,moves,abilities,type-chart,items,affixes,skill-trees,monsters,dungeons,loot-tables,move-behaviors}.json`。
- 驗證：技能樹 move 節點 ∈ learnset；每個 learnset 中被樹引用的招式都有 MoveBehavior；所有參照 id 都存在。驗證失敗時 exit 1。
- 種子：`/workspace/starters/data/*.json` 有已抓好的御三家資料，可複製到新 repo 的 `data/raw/seed/` 當離線備援。
